<script>
let currentUser = null;
let notes = [];
let currentId = null;
let saveTimer = null;
// 全局提示工具
function toast(msg) {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.classList.add('show');
  setTimeout(() => el.classList.remove('show'), 2000);
}
// 页面切换
function showPage(id) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.getElementById(id).classList.add('active');
  window.scrollTo(0, 0);
}
// 账号管理工具
function getAllUsers() {
  return JSON.parse(localStorage.getItem('notepad_users') || '[]');
}
function saveUsers(users) {
  localStorage.setItem('notepad_users', JSON.stringify(users));
}
// 注册逻辑
function doRegister() {
  const phone = document.getElementById('regPhone').value.trim();
  const pwd = document.getElementById('regPwd').value;
  const pwd2 = document.getElementById('regPwd2').value;
  if (!/^1\d{10}$/.test(phone)) return toast('请输入正确的11位手机号');
  if (pwd.length < 6) return toast('密码长度不能少于6位');
  if (pwd !== pwd2) return toast('两次输入的密码不一致');
  const users = getAllUsers();
  if (users.find(u => u.phone === phone)) return toast('该手机号已经注册过了');
  users.push({
    phone,
    password: btoa(pwd),
    notes: [{
      id: Date.now(),
      title: '欢迎使用在线记事本',
      content: '· 这里只显示标题，点卡片进入全屏编辑\n· 打字自动保存，多设备登录即可同步\n· 你的笔记会和你的账号绑定',
      updated: Date.now()
    }]
  });
  saveUsers(users);
  toast('注册成功，请登录');
  setTimeout(() => showPage('pageLogin'), 800);
}
// 登录逻辑
function doLogin() {
  const phone = document.getElementById('loginPhone').value.trim();
  const pwd = document.getElementById('loginPwd').value;
  const users = getAllUsers();
  const user = users.find(u => u.phone === phone && u.password === btoa(pwd));
  if (!user) return toast('手机号或密码错误');
  currentUser = user;
  localStorage.setItem('notepad_current', phone);
  notes = user.notes;
  toast('登录成功');
  afterLogin();
}
// 自动登录检测
function autoLogin() {
  const currentPhone = localStorage.getItem('notepad_current');
  if (!currentPhone) return false;
  const users = getAllUsers();
  const user = users.find(u => u.phone === currentPhone);
  if (!user) return false;
  currentUser = user;
  notes = user.notes;
  return true;
}
// 登录完成后初始化列表
function afterLogin() {
  document.getElementById('userPhone').textContent = currentUser.phone.replace(/(\d{3})\d{4}(\d{4})/, '$1****$2');
  renderList();
  showPage('pageList');
}
// 退出登录
function doLogout() {
  if (!confirm('确定要退出登录吗？')) return;
  currentUser = null;
  notes = [];
  currentId = null;
  localStorage.removeItem('notepad_current');
  showPage('pageLogin');
  toast('已退出登录');
}
// 保存笔记到用户账号
function save() {
  if (!currentUser) return;
  currentUser.notes = notes;
  const users = getAllUsers().map(u => u.phone === currentUser.phone ? currentUser : u);
  saveUsers(users);
}
function current() {
  return notes.find(n => n.id === currentId);
}
function fmtTime(ts) {
  const d = new Date(ts);
  return d.toLocaleString('zh-CN', { month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit' });
}
/* ===== 列表页 ===== */
function renderList() {
  const keyword = document.getElementById('searchInput').value.trim();
  const list = notes.filter(n => (n.title + '\n' + n.content).includes(keyword));
  document.getElementById('emptyTip').hidden = list.length > 0;
  document.getElementById('noteList').innerHTML = list.map(n => `
    <div class="note-card" onclick="openNote(${n.id})">
      <div class="info">
        <div class="title"></div>
        <div class="date">${fmtTime(n.updated)}</div>
      </div>
      <button class="del" onclick="event.stopPropagation(); deleteNote(${n.id})">删除</button>
    </div>
  `).join('');
  document.querySelectorAll('#noteList .title').forEach((el, i) => {
    el.textContent = list[i].title || '（无标题）';
  });
}
/* ===== 编辑页 ===== */
function openNote(id) {
  currentId = id;
  const n = current();
  document.getElementById('titleInput').value = n.title;
  document.getElementById('contentInput').value = n.content;
  document.getElementById('saveStatus').textContent = '已保存 ' + fmtTime(n.updated);
  document.getElementById('syncStatus').textContent = '已同步';
  updateCount();
  showPage('pageEdit');
  setTimeout(() => document.getElementById('contentInput').focus(), 100);
}
function newNote() {
  const n = { id: Date.now(), title: '', content: '', updated: Date.now() };
  notes.unshift(n);
  save();
  renderList();
  openNote(n.id);
}
function backToList() {
  clearTimeout(saveTimer);
  doSave();
  currentId = null;
  renderList();
  showPage('pageList');
}
function deleteNote(id) {
  if (!confirm('确定删除这篇笔记吗？')) return;
  notes = notes.filter(n => n.id !== id);
  save();
  renderList();
}
function delNote() {
  if (notes.length <= 1) return alert('至少保留一篇笔记！');
  if (!confirm('确定删除这篇笔记吗？')) return;
  deleteNote(currentId);
  backToList();
}
function updateCount() {
  document.getElementById('charCount').textContent =
    document.getElementById('contentInput').value.length + ' 字';
}
function doSave() {
  const n = current();
  if (!n) return;
  n.title = document.getElementById('titleInput').value.trim();
  n.content = document.getElementById('contentInput').value;
  n.updated = Date.now();
  save();
  document.getElementById('saveStatus').textContent = '已保存 ' + fmtTime(n.updated);
  document.getElementById('syncStatus').textContent = '已同步';
  updateCount();
}
function scheduleSave() {
  document.getElementById('saveStatus').textContent = '保存中…';
  document.getElementById('syncStatus').textContent = '同步中…';
  clearTimeout(saveTimer);
  saveTimer = setTimeout(doSave, 500);
}
document.getElementById('titleInput').addEventListener('input', scheduleSave);
document.getElementById('contentInput').addEventListener('input', scheduleSave);
document.getElementById('titleInput').addEventListener('keydown', e => {
  if (e.key === 'Enter') { e.preventDefault(); document.getElementById('contentInput').focus(); }
});
// 初始化
if (autoLogin()) {
  afterLogin();
} else {
  showPage('pageLogin');
}
</script>
